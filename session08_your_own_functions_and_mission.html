Chunk 2 of 2

A program that reads like a plan

This program has two functions of its own: follow_to_bar is the line follower from session 7, and quarter_turn is the measured turn.

With the hard parts named, main is short. Lines 34 to 37 are the plan, one step on each line. You can read what the robot will do without reading how it does it.

from hub import motion_sensor, port, sound
import color_sensor
import motor_pair
import runloop

LINE_SENSOR = port.B
STOP_SENSOR = port.F
DARK = 50
FORWARD = 360
QUARTER = 900


async def follow_to_bar():
    while color_sensor.reflection(STOP_SENSOR) > DARK:
        if color_sensor.reflection(LINE_SENSOR) < DARK:
            motor_pair.move(motor_pair.PAIR_1, 30, velocity=150)
        else:
            motor_pair.move(motor_pair.PAIR_1, -30, velocity=150)
    motor_pair.stop(motor_pair.PAIR_1)


async def quarter_turn():
    motion_sensor.reset_yaw(0)
    await runloop.sleep_ms(100)
    motor_pair.move(motor_pair.PAIR_1, 100, velocity=100)
    await runloop.until(lambda: abs(motion_sensor.tilt_angles()[0]) > QUARTER)
    motor_pair.stop(motor_pair.PAIR_1)


async def main():
    # The left drive motor is in port A. The right drive motor is in port E.
    motor_pair.pair(motor_pair.PAIR_1, port.A, port.E)

    await follow_to_bar()
    await quarter_turn()
    await motor_pair.move_for_degrees(motor_pair.PAIR_1, FORWARD, 0, velocity=250)
    await sound.beep(880, 500, 100)


runloop.run(main())
